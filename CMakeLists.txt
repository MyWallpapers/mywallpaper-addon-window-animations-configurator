cmake_minimum_required(VERSION 3.24)
project(MyWallpaperWindowAnimations VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(animation_model STATIC src/config.cpp src/motion.cpp src/render.cpp)
target_include_directories(animation_model PUBLIC include third_party)
if(MSVC)
  target_compile_options(animation_model PUBLIC /utf-8 /W4 /permissive- /EHsc /Zc:preprocessor /Brepro)
else()
  target_compile_options(animation_model PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(animation_contract_tests tests/contract_tests.cpp)
target_link_libraries(animation_contract_tests PRIVATE animation_model)
enable_testing()
add_test(NAME animation_contract COMMAND animation_contract_tests "${CMAKE_SOURCE_DIR}")
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 is required")
  endif()
  add_executable(window_animations src/companion.cpp src/runtime.cpp src/watchdog.cpp src/capture.cpp)
  target_compile_definitions(window_animations PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(window_animations PRIVATE animation_model user32 gdi32 dwmapi)
  set_target_properties(window_animations PROPERTIES OUTPUT_NAME companion)
  add_executable(animation_preview WIN32 src/preview.cpp)
  if(MSVC)
    target_compile_options(animation_preview PRIVATE /utf-8 /W4 /permissive- /EHsc /Brepro)
    target_link_options(window_animations PRIVATE /Brepro)
    target_link_options(animation_preview PRIVATE /Brepro)
  endif()
  target_compile_definitions(animation_preview PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(animation_preview PRIVATE user32 gdi32 dwmapi)
  set_target_properties(animation_preview PROPERTIES OUTPUT_NAME window-animations-preview)
endif()
